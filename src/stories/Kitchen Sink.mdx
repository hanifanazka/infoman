## Menu

{(() => {
  const actions = [
    {
      label: "Hi",
      onClick: () => alert("Hi"),
    },
    {
      label: "Loognsggsg",
      onClick: () => alert("Loognsggsg"),
    },
  ];
  const menuAnchor = (anchor) => 
    <Menu {...{
      anchor,
      actions,
      children: "Menu",
    }}/>
  return <div style={{ display: "flex", alignItems: "center", gap: 22 }}>
    <div style={{ display: "grid", gridTemplateColumns: "repeat(3, 1fr)", width: 400, gap: 4 }}>
      {[
        "bottom start",
        "bottom center",
        "bottom end",
        "top start",
        "top center",
        "top end",
      ].map((anchor) => (
      <div>
        <p style={{ textWrap: "nowrap" }}>{anchor}</p>
        {menuAnchor(anchor)}
      </div>
      ))}
    </div>
    <div style={{ display: "grid", gridTemplateColumns: "repeat(2, 1fr)", width: 200, gap: 4 }}>
      {[
    "right start",
    "left start",
    "right center",
    "left center",
    "right end",
    "left end",
      ].map((anchor) => (
      <div>
        <p style={{ textWrap: "nowrap" }}>{anchor}</p>
        {menuAnchor(anchor)}
      </div>
      ))}
    </div>
    <div style={{ display: "grid", gridTemplateColumns: "repeat(2, 1fr)", width: 200, gap: 4 }}>
      <div>
        <p>Menu icon</p>
        <Menu anchor="bottom center" {...{actions}}>
          <UserCircleIcon style={{ width: 24, height: 24, display: "block" }} />
        </Menu>
      </div>
      <div>
        <p>smaller</p>
        <Menu anchor="bottom center" size="sm" {...{actions}}>
          <UserCircleIcon style={{ width: 24, height: 24, display: "block" }} />
        </Menu>
      </div>
      <div>
        <p>secondary</p>
        <Menu anchor="bottom center" size="sm" color="secondary" {...{actions}}>
          <UserCircleIcon style={{ width: 24, height: 24, display: "block" }} />
        </Menu>
      </div>
      <div>
        <p>plain</p>
        <Menu anchor="bottom center" size="sm" color="secondary" variant="plain" {...{actions}}>
          <UserCircleIcon style={{ width: 24, height: 24, display: "block" }} />
        </Menu>
      </div>
    </div>
  </div>;
})()}
